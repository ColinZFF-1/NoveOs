## 3. 状态管理（Zustand）

### 3.1 全局 Store 设计

```typescript
// stores/novelOSStore.ts
import { create } from 'zustand';
import { devtools, persist } from 'zustand/middleware';

interface NovelOSState {
  // === 全局数据 ===
  projects: Project[];
  systemStats: SystemStats | null;

  // === 当前项目数据 ===
  currentProjectId: string | null;
  currentProject: Project | null;
  pipeline: Pipeline | null;
  chapters: ChapterSummary[];
  currentChapter: ChapterDetail | null;
  characters: Character[];
  emotions: EmotionPoint[];
  logs: LogPayload[];

  // === UI 状态 ===
  isLoading: boolean;
  error: string | null;
  wsConnected: boolean;
  sidebarCollapsed: boolean;

  // === Actions ===
  // 项目
  fetchProjects: () => Promise<void>;
  selectProject: (id: string) => Promise<void>;
  createProject: (data: CreateProjectDTO) => Promise<void>;

  // 流水线
  startPipeline: (range?: string) => Promise<void>;
  pausePipeline: () => Promise<void>;
  updatePipeline: (pipeline: Pipeline) => void;

  // 章节
  fetchChapters: () => Promise<void>;
  fetchChapter: (num: number) => Promise<void>;
  completeChapter: (chapter: ChapterCompletePayload) => void;

  // 角色
  fetchCharacters: () => Promise<void>;

  // 情绪
  fetchEmotions: () => Promise<void>;

  // 日志
  addLog: (log: LogPayload) => void;
  clearLogs: () => void;

  // WebSocket
  connectWebSocket: () => void;
  disconnectWebSocket: () => void;
  setWsConnected: (connected: boolean) => void;

  // 错误
  setError: (error: string | null) => void;
  clearError: () => void;
}

export const useNovelOSStore = create<NovelOSState>()(
  devtools(
    persist(
      (set, get) => ({
        // 初始状态
        projects: [],
        systemStats: null,
        currentProjectId: null,
        currentProject: null,
        pipeline: null,
        chapters: [],
        currentChapter: null,
        characters: [],
        emotions: [],
        logs: [],
        isLoading: false,
        error: null,
        wsConnected: false,
        sidebarCollapsed: false,

        // Actions 实现
        fetchProjects: async () => {
          set({ isLoading: true });
          try {
            const res = await apiClient.get('/projects');
            set({ projects: res.data.items, isLoading: false });
          } catch (e) {
            set({ error: '加载项目失败', isLoading: false });
          }
        },

        selectProject: async (id) => {
          set({ isLoading: true, currentProjectId: id });
          try {
            const [project, pipeline, chapters, characters, emotions] = await Promise.all([
              apiClient.get(`/projects/${id}`),
              apiClient.get(`/projects/${id}/pipeline`),
              apiClient.get(`/projects/${id}/chapters`),
              apiClient.get(`/projects/${id}/characters`),
              apiClient.get(`/projects/${id}/emotions`),
            ]);
            set({
              currentProject: project.data,
              pipeline: pipeline.data,
              chapters: chapters.data.items,
              characters: characters.data.items,
              emotions: emotions.data.items,
              isLoading: false,
            });
            get().connectWebSocket();
          } catch (e) {
            set({ error: '加载项目详情失败', isLoading: false });
          }
        },

        startPipeline: async (range) => {
          const { currentProjectId } = get();
          if (!currentProjectId) return;
          try {
            await apiClient.post(`/projects/${currentProjectId}/pipeline/start`, {
              resume: true,
              chapter_range: range,
            });
          } catch (e: any) {
            if (e.response?.status === 409) {
              set({ error: '项目正在运行中，请先暂停' });
            }
          }
        },

        addLog: (log) => {
          set((state) => ({
            logs: [log, ...state.logs].slice(0, 5000), // 最多保留 5000 条
          }));
        },

        connectWebSocket: () => {
          const { currentProjectId, wsConnected } = get();
          if (!currentProjectId || wsConnected) return;

          const ws = new WebSocket(`ws://localhost:8000/ws/projects/${currentProjectId}`);
          ws.onopen = () => set({ wsConnected: true });
          ws.onmessage = (event) => {
            const msg: WSMessage = JSON.parse(event.data);
            switch (msg.type) {
              case 'log':
                get().addLog(msg.payload);
                break;
              case 'progress':
                // 更新流程图进度
                break;
              case 'pipeline_update':
                set({ pipeline: msg.payload });
                break;
              case 'chapter_complete':
                get().completeChapter(msg.payload);
                break;
              case 'system_status':
                // 更新底部状态栏
                break;
            }
          };
          ws.onclose = () => set({ wsConnected: false });
          ws.onerror = () => set({ wsConnected: false, error: 'WebSocket 连接中断' });
        },

        // ... 其他 actions
      }),
      {
        name: 'novel-os-store',
        partialize: (state) => ({ 
          currentProjectId: state.currentProjectId,
          sidebarCollapsed: state.sidebarCollapsed,
        }), // 只持久化部分字段
      }
    )
  )
);
```

### 5.2 加载状态矩阵

| 场景 | 加载样式 | 占位内容 | 错误处理 |
|------|---------|---------|---------|
| 项目列表首次加载 | 骨架屏（3 行） | - | 空状态 + 重试按钮 |
| 切换项目 | 全屏 Loading 遮罩 | 保留旧数据 | Toast 错误提示 |
| 章节内容加载 | 打字机效果 | - | 显示错误边界 |
| 流程图推进 | 节点呼吸灯 | 保持旧状态 | 节点变红 + 错误提示 |
| 日志流 | 底部 Loading 指示器 | 保持历史日志 | 显示"连接中断"横幅 |
| 情绪曲线 | 图表淡入 | - | 显示空图表 |

### 5.3 关键交互流程

**一键启动流程**
```
用户点击 LaunchButton
  -> 按钮缩放动画 (scale 0.95) + 火箭抖动
  -> 调用 store.startPipeline()
  -> API POST /pipeline/start
  -> 成功: PipelineFlow 当前节点变 running，开始呼吸动画
  -> 失败 (409): Toast "项目正在运行中"
  -> 失败 (500): Toast "启动失败，请重试"
```

**章节切换流程**
```
用户点击章节目录第 N 章
  -> 当前章节高亮
  -> 调用 store.fetchChapter(N)
  -> API GET /chapters/N
  -> 成功: ChapterPreview 内容清空，打字机重新播放
  -> 成功: EmotionCurve 当前点标记移动
  -> 成功: AuditPanel 数据更新
  -> 失败: 显示错误边界 "章节加载失败"
```

**WebSocket 断线重连**
```
连接断开
  -> 显示顶部横幅 "实时连接中断，写作仍在后台进行"
  -> 启动轮询: 每 5 秒 GET /status
  -> 重连成功
    -> 横幅变为绿色 "实时连接已恢复"
    -> 3 秒后横幅消失
    -> 停止轮询，恢复 WebSocket
  -> 重连失败 (60 秒后)
    -> 横幅变为红色 "连接异常，建议刷新页面"
    -> 显示 "刷新" 按钮
```

## 6. 与后端对接

### 6.1 Axios 客户端配置

```typescript
// api/client.ts
import axios from 'axios';

export const apiClient = axios.create({
  baseURL: import.meta.env.VITE_API_URL || 'http://localhost:8000/api/v1',
  timeout: 30000,
  headers: {
    'Content-Type': 'application/json',
  },
});

// 请求拦截器
apiClient.interceptors.request.use(
  (config) => {
    const token = localStorage.getItem('token');
    if (token) config.headers.Authorization = `Bearer ${token}`;
    return config;
  },
  (error) => Promise.reject(error)
);

// 响应拦截器
apiClient.interceptors.response.use(
  (response) => response.data, // 直接返回 data（后端包装格式）
  (error) => {
    if (error.response?.status === 429) {
      // 限流提示
      return Promise.reject(new Error('请求过于频繁，请稍后再试'));
    }
    if (error.response?.status >= 500) {
      return Promise.reject(new Error('服务器异常，请联系管理员'));
    }
    return Promise.reject(error);
  }
);
```

```typescript
import { useEffect, useRef, useCallback } from 'react';
import { useNovelOSStore } from '@/stores/novelOSStore';

export function useWebSocket(projectId: string | null) {
  const wsRef = useRef<WebSocket | null>(null);
  const reconnectTimer = useRef<NodeJS.Timeout>();
  const reconnectCount = useRef(0);
  const MAX_RECONNECT = 5;

  const setWsConnected = useNovelOSStore((s) => s.setWsConnected);
  const addLog = useNovelOSStore((s) => s.addLog);
  const updatePipeline = useNovelOSStore((s) => s.updatePipeline);

  const connect = useCallback(() => {
    if (!projectId) return;

    const ws = new WebSocket(`ws://localhost:8000/ws/projects/${projectId}`);
    wsRef.current = ws;

    ws.onopen = () => {
      setWsConnected(true);
      reconnectCount.current = 0;
    };

    ws.onmessage = (event) => {
      const msg = JSON.parse(event.data);
      handleMessage(msg);
    };

    ws.onclose = () => {
      setWsConnected(false);
      if (reconnectCount.current < MAX_RECONNECT) {
        reconnectTimer.current = setTimeout(() => {
          reconnectCount.current++;
          connect();
        }, 3000);
      }
    };

    ws.onerror = () => {
      ws.close();
    };
  }, [projectId]);

  const handleMessage = (msg: WSMessage) => {
    switch (msg.type) {
      case 'log':
        addLog(msg.payload);
        break;
      case 'pipeline_update':
        updatePipeline(msg.payload);
        break;
      // ... 其他 case
    }
  };

  useEffect(() => {
    connect();
    return () => {
      wsRef.current?.close();
      clearTimeout(reconnectTimer.current);
    };
  }, [connect]);

  return { send: (data: object) => wsRef.current?.send(JSON.stringify(data)) };
}
```

## 7. 性能优化

### 7.1 渲染优化

| 技术 | 应用场景 | 实现方式 |
|------|---------|---------|
| `React.memo` | CharacterCard、LogLine | 纯展示组件，props 不变不渲染 |
| `useMemo` | 流程图进度计算、情绪曲线数据转换 | 缓存计算结果 |
| `useCallback` | 事件处理器传递给子组件 | 避免函数引用变化 |
| `虚拟滚动` | 日志流（5000+ 条）、章节目录（240 章） | react-virtuoso |
| `懒加载` | 路由级组件 | React.lazy + Suspense |
| `图片懒加载` | 角色头像 | loading="lazy" |
| `will-change` | 流程图呼吸动画节点 | CSS will-change: transform |

### 7.2 数据优化

| 策略 | 实现 | 效果 |
|------|------|------|
| 日志截断 | 只保留最近 5000 条，旧数据归档 | 内存占用稳定 |
| 章节分页 | 目录树每页 20 章 | 减少首次加载 |
| 情绪曲线采样 | 超过 100 章时数据点合并 | ECharts 渲染不卡顿 |
| 正文分段加载 | 先加载前 3000 字，滚动到底再加载剩余 | 首屏 < 1s |
| WebSocket 心跳 | 30s ping/pong | 及时感知断线 |
| 请求去重 | 相同 API 并发时只发一个 | 减少服务器压力 |

## 8. 错误处理与降级

### 8.1 错误边界（Error Boundary）

```typescript
// components/common/ErrorBoundary.tsx
import { Component, ReactNode } from 'react';

interface Props {
  children: ReactNode;
  fallback?: ReactNode;
}

interface State {
  hasError: boolean;
  error?: Error;
}

export class ErrorBoundary extends Component<Props, State> {
  state: State = { hasError: false };

  static getDerivedStateFromError(error: Error): State {
    return { hasError: true, error };
  }

  componentDidCatch(error: Error, errorInfo: React.ErrorInfo) {
    console.error('ErrorBoundary caught:', error, errorInfo);
    // 上报 Sentry
  }

  render() {
    if (this.state.hasError) {
      return this.props.fallback || (
        <div className="p-8 text-center">
          <h2 className="text-lg font-bold text-red-500 mb-2">组件渲染异常</h2>
          <p className="text-sm text-gray-500 mb-4">{this.state.error?.message}</p>
          <button 
            onClick={() => window.location.reload()}
            className="px-4 py-2 bg-primary-500 text-white rounded-lg"
          >
            刷新页面
          </button>
        </div>
      );
    }
    return this.props.children;
  }
}
```

### 8.2 降级策略矩阵

| 场景 | 正常状态 | 降级状态 | 恢复策略 |
|------|---------|---------|---------|
| WebSocket 连接 | 实时推送日志 | 轮询 GET /status 每 5s | 自动重连成功后恢复 |
| ECharts 图表 | 交互式图表 | 静态 SVG 简图 | 库加载成功后恢复 |
| 角色头像 | 像素风 SVG | 首字母占位 | 网络恢复后重新加载 |
| 章节正文 | 完整内容 | 前 1000 字 + "加载更多" | 点击后加载剩余 |
| 流程图动画 | CSS 呼吸动画 | 静态颜色标记 | 浏览器支持检测后恢复 |
| 项目列表 | 实时数据 | 本地缓存数据 + 刷新按钮 | 网络恢复后自动刷新 |
